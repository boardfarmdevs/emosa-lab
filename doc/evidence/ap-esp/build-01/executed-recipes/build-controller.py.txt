"""Build an isolated native counter-capability candidate; never install in a peer.

Run on an x86_64 Ubuntu 22.04 HOST with g++, make, patch, git, strip and cmake.
The verified source and install archives contain the original 22 lab patches.
By default only controller sources are rebuilt. --lifecycle additionally rebuilds
the selected Linux BPL library and checks its destruction order against baseline.
"""

import argparse
import hashlib
import json
import os
import shutil
import subprocess
import tarfile
from pathlib import Path

HERE = Path(__file__).resolve().parent
ARCHIVES = {
    "prplmesh-patched-source.tar.gz": (
        "ebc771d43279cdb6b1cf5e2c09eb51d66ef3a8afab6b97f4561439d2606ac549"
    ),
    "prpl-install-nl80211-6.0.0.tar.gz": (
        "a98a6903fb9db1494ece0c292dd85c7d4d8ff2f570b5b85768ed6f299f72fb1f"
    ),
    "prpl-runtime-deps-6.0.0.tar.gz": (
        "ad912dc63b7d2188096bd6c0a42978590c4a4b82b0590da1198514e122cec4d2"
    ),
}


def digest(path):
    with path.open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--build", type=Path, required=True, help="New private absolute directory")
    parser.add_argument("--artifacts", type=Path, required=True)
    parser.add_argument("--jobs", type=int, choices=range(1, 9), default=3)
    parser.add_argument(
        "--onboarding",
        action="store_true",
        help="Also omit unsupported/empty configuration companions",
    )
    parser.add_argument("--lifecycle", action="store_true", help="Fix native BPL model lifetime")
    parser.add_argument(
        "--ap-esp", action="store_true", help="Validate and decode sparse AP ESP arrays"
    )
    args = parser.parse_args()
    if not args.build.is_absolute() or args.build.exists():
        parser.error("Build directory must be new and absolute")
    for name, expected in ARCHIVES.items():
        if digest(args.artifacts / name) != expected:
            raise SystemExit(f"Archive digest mismatch: {name}")
    os.umask(0o077)
    args.build.mkdir(parents=True, exist_ok=False)
    source, install = args.build / "source", args.build / "prpl-install-nl80211"
    source.mkdir()
    # These exact, locally reviewed archives have already passed their hash checks.
    with tarfile.open(args.artifacts / "prplmesh-patched-source.tar.gz") as archive:
        # The upstream source archive carries an unused parent-directory toolchain
        # symlink. The overlay uses the host compiler, not that external config.
        archive.extractall(
            source,
            members=[m for m in archive if m.name != "./external_toolchain.cfg"],
            filter="data",
        )
    with tarfile.open(args.artifacts / "prpl-install-nl80211-6.0.0.tar.gz") as archive:
        archive.extractall(
            args.build,
            members=[
                m
                for m in archive
                if m.name.startswith(("prpl-install-nl80211/include/", "prpl-install-nl80211/lib/"))
            ],
            filter="data",
        )
    with tarfile.open(args.artifacts / "prpl-runtime-deps-6.0.0.tar.gz") as archive:
        archive.extractall(args.build, filter="data")
    headers = args.build / "headers"
    headers.mkdir()
    lock = json.loads((HERE / "controller-headers.json").read_text())
    with (args.build / "build.log").open("w") as log:

        def run(*command, **kwargs):
            return subprocess.run(
                command, check=True, stdout=log, stderr=log, timeout=900, **kwargs
            )

        for row in lock["libraries"]:
            target = headers / row["name"]
            run("git", "init", "--quiet", str(target))
            run(
                "git",
                "-C",
                str(target),
                "fetch",
                "--quiet",
                "--depth=1",
                row["repository"],
                row["commit"],
            )
            run("git", "-C", str(target), "checkout", "--quiet", "--detach", "FETCH_HEAD")
            actual = subprocess.check_output(
                ["git", "-C", str(target), "rev-parse", "HEAD"], text=True
            )
            if actual.strip() != row["commit"]:
                raise RuntimeError("Dependency checkout differs from header lock")
        patch = HERE.parent / "patches/0004-controller-counter-capability.patch"
        with patch.open() as content:
            run("patch", "--batch", "--fuzz=0", "-p1", "-d", str(source), stdin=content)
        extra_patches = []
        if args.onboarding:
            extra = HERE.parent / "patches/0005-controller-configuration-scope.patch"
            with extra.open() as content:
                run("patch", "--batch", "--fuzz=0", "-p1", "-d", str(source), stdin=content)
            extra_patches.append({"name": extra.name, "sha256": digest(extra)})
        if args.lifecycle:
            extra = HERE.parent / "patches/0006-bpl-model-lifetime.patch"
            with extra.open() as content:
                run("patch", "--batch", "--fuzz=0", "-p1", "-d", str(source), stdin=content)
            extra_patches.append({"name": extra.name, "sha256": digest(extra)})
        if args.ap_esp:
            extra = HERE.parent / "patches/0007-controller-ap-esp-presence.patch"
            with extra.open() as content:
                run("patch", "--batch", "--fuzz=0", "-p1", "-d", str(source), stdin=content)
            extra_patches.append({"name": extra.name, "sha256": digest(extra)})
        run(
            "cmake",
            "-S",
            str(HERE.parent / "controller-overlay"),
            "-B",
            str(args.build / "build"),
            "-DCMAKE_BUILD_TYPE=RelWithDebInfo",
            f"-DPEER_SOURCE={source}",
            f"-DPEER_INSTALL={install}",
            f"-DHEADERS={headers}",
            f"-DRUNTIME={args.build}",
            f"-DEMOSA_LIFECYCLE={'ON' if args.lifecycle else 'OFF'}",
            f"-DEMOSA_AP_ESP={'ON' if args.ap_esp else 'OFF'}",
            "-DCMAKE_INSTALL_LIBDIR=lib",
        )
        run("cmake", "--build", str(args.build / "build"), "--parallel", str(args.jobs))
        run("cmake", "--install", str(args.build / "build"), "--prefix", str(args.build / "stage"))
        candidate = args.build / "stage/bin/beerocks_controller"
        unstripped_sha = digest(candidate)
        run("strip", "--strip-debug", str(candidate))
        environment = os.environ | {
            "LD_LIBRARY_PATH": ":".join(
                str(p)
                for p in (
                    install / "lib",
                    args.build / "usr/lib/x86_64-linux-gnu",
                    args.build / "usr/lib",
                )
            )
        }
        regression = subprocess.check_output(
            [str(args.build / "build/emosa-counter-regression")],
            env=environment,
            text=True,
            timeout=30,
        )
        (args.build / "regression.json").write_text(regression)
        if json.loads(regression)["passed"] != 12:
            raise RuntimeError("Native counter regression did not pass")
        ap_esp = None
        if args.ap_esp:
            raw = subprocess.check_output(
                [str(args.build / "build/emosa-ap-esp-regression")],
                env=environment,
                text=True,
                timeout=30,
            )
            (args.build / "ap-esp-regression.json").write_text(raw)
            ap_esp = json.loads(raw)
            if ap_esp["passed"] != 240 or not ap_esp["baseline_fixed_vi_offset_returns_null"]:
                raise RuntimeError("Native AP ESP regression did not pass")
        runtime_libraries, lifetime = {}, None
        if args.lifecycle:
            name = "libbpl.so.6.0.0"
            library = args.build / "stage/lib" / name
            unstripped_library = digest(library)
            run("strip", "--strip-debug", str(library))
            runtime_libraries[name] = {
                "baseline_sha256": digest(install / "lib" / name),
                "candidate_sha256": digest(library),
                "unstripped_sha256": unstripped_library,
            }
            lifetime = {}
            for variant, first in (("baseline", install / "lib"), ("candidate", library.parent)):
                env = environment | {
                    "LD_LIBRARY_PATH": str(first) + ":" + environment["LD_LIBRARY_PATH"]
                }
                program = str(args.build / "build/emosa-model-lifetime")
                dependencies = subprocess.check_output(["ldd", program], env=env, text=True)
                wanted_library = str((first / name).resolve())
                loaded = next(
                    line.split()[2]
                    for line in dependencies.splitlines()
                    if "libbpl.so.6 =>" in line
                )
                if str(Path(loaded).resolve()) != wanted_library:
                    raise RuntimeError("Lifecycle probe did not load the selected library")
                outputs = {}
                for case, arguments in (("implicit", []), ("explicit_clear", ["--clear"])):
                    outputs[case] = subprocess.check_output(
                        [program, *arguments], env=env, text=True, timeout=30
                    ).splitlines()
                lifetime[variant] = {
                    "loaded_library_sha256": digest(first / name),
                    "observations": outputs,
                }
            if lifetime["baseline"]["observations"]["implicit"] != [
                "main_returning",
                "runtime_destroyed",
                "model_destroyed",
            ]:
                raise RuntimeError("Baseline lifetime defect was not reproduced")
            if lifetime["candidate"]["observations"]["implicit"] != [
                "main_returning",
                "model_destroyed",
                "runtime_destroyed",
            ]:
                raise RuntimeError("Candidate model outlived runtime")
            for row in lifetime.values():
                if row["observations"]["explicit_clear"] != [
                    "model_destroyed",
                    "main_returning",
                    "runtime_destroyed",
                ]:
                    raise RuntimeError("Explicit BPL release changed behavior")
            (args.build / "lifetime-regression.json").write_text(
                json.dumps(lifetime, indent=2) + "\n"
            )
    report = {
        "scope": "Native controller counter advertisement candidate; not admitted onboarding",
        "input_archives": ARCHIVES,
        "header_dependencies": lock,
        "patch_sha256": digest(patch),
        "extra_patches": extra_patches,
        "runtime_libraries": runtime_libraries,
        "lifetime_regression": lifetime,
        "ap_esp_regression": ap_esp,
        "candidate_sha256": digest(candidate),
        "unstripped_candidate_sha256": unstripped_sha,
        "source_sha256": {
            p: digest(source / p)
            for p in (
                "controller/src/beerocks/master/controller.cpp",
                *(["controller/src/beerocks/master/ap_metrics_esp.h"] if args.ap_esp else []),
                "controller/src/beerocks/master/db/db.cpp",
                "controller/src/beerocks/master/tasks/agent_monitoring_task.cpp",
                *(
                    ("framework/platform/bpl/cfg/dm/" + name)
                    for name in ("bpl_amx.cpp", "bpl_amb_ptr.h", "bpl_cfg_amx_helper.h", "bpl.cpp")
                    if args.lifecycle
                ),
            )
        },
        "recipe_sha256": {
            p.name: digest(p)
            for p in (
                Path(__file__),
                HERE / "counter-units.cpp",
                HERE / "controller-headers.json",
                HERE.parent / "controller-overlay/CMakeLists.txt",
                *([HERE / "model-lifetime.cpp"] if args.lifecycle else []),
                *([HERE / "ap-esp.cpp"] if args.ap_esp else []),
            )
        },
        "tools": {
            name: subprocess.check_output([name, "--version"], text=True).splitlines()[0]
            for name in ("cmake", "g++", "git", "strip")
        },
        "counter_regression": json.loads(regression),
        "installed_in_peer": False,
    }
    (args.build / "candidate.json").write_text(json.dumps(report, indent=2) + "\n")
    shutil.copyfile(HERE / "controller-headers.json", args.build / "controller-headers.json")
    print(json.dumps(report, indent=2))


if __name__ == "__main__":
    main()
